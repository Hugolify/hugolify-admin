{{- $collection := . -}} {{- $block := print "selected-" . -}} {{- $section := dict "section-hidden" $collection -}} {{- $layouts := partial "admin/func/GetLayouts.html" (slice "carousel" "grid" "list") -}} {{- $uiFields := site.Params.admin.fields.ui.fields | append (dict "layout" $layouts) "scrollsnap" -}} {{- if in $layouts "carousel" -}} {{- $uiFields = $uiFields | append "carousel" -}} {{- end -}} {{- $ui := dict "ui" $uiFields -}} {{/* Taxonomies of the section, used to filter the pages when source is "taxonomies" */}} {{- $taxonomies := partial "admin/func/GetTaxonomies.html" $collection -}} {{/* A nested collection holds a tree, so a block can take the children of one of its pages, the sub-expertises of an expertise for one, or the first level of the tree: the sources "children" and "toplevel" are offered to those collections only */}} {{- $nested := gt (partial "admin/func/GetNestedDepth.html" $collection) 1 -}} {{/* Sources the section can offer, the manual selection always being one of them */}} {{- $sources := slice "selection" -}} {{- if $taxonomies -}} {{- $sources = $sources | append "taxonomies" -}} {{- end -}} {{- if $nested -}} {{- $sources = $sources | append "children" "toplevel" -}} {{- end -}} {{/* Without another source, the manual selection is the only one, so it is required */}} {{- $relation := dict "relation" (dict "collection" $collection "name" "items" "multiple" true "required" (eq (len $sources) 1)) -}} {{- $fields := slice "heading" $section -}} {{- if gt (len $sources) 1 -}} {{/* Fields shown for each source, the CMS that can hide the others;
